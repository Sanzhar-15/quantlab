## Objective

Implement the complete approval flow for file changes, properly integrating the ApprovalToken security mechanism. This addresses the corrected GAP-05 understanding.

---

## Context

From the corrected audit (GAP-05):
- Host generates the editScriptHash, not webview
- Webview just echoes back the hash with approval decision
- Host creates ApprovalToken internally after approval

The flow:
1. Host sends `diff-preview` with `editScriptHash`
2. User clicks Accept/Reject in webview
3. Webview sends `changes:accept` or `changes:reject` with hash
4. Host validates hash, creates ApprovalToken, applies changes

Reference: `QIC_UI_SPEC/Optimal_plan/06-CHANGES-AND-DIFF.md`

### 1. Create Approval Manager

```javascript
// src/vs/workbench/contrib/qic/browser/media/approvalManager.js
// @ts-nocheck
/**
 * QIC Approval Manager
 * Handles change approval/rejection flow
 * GAP-05 FIX: Proper hash echoing
 */

(function() {
    'use strict';

    // ═══════════════════════════════════════════════════════════════════
    // State
    // ═══════════════════════════════════════════════════════════════════

    // Map of changeId -> editScriptHash (sent by host)
    const changeHashes = new Map();

    // Currently expanded change card
    let expandedChangeId = null;

    // ═══════════════════════════════════════════════════════════════════
    // Public API
    // ═══════════════════════════════════════════════════════════════════

    /**
     * Register a change with its hash (called when host sends diff-preview)
     * @param {string} changeSetId
     * @param {string} changeId
     * @param {string} editScriptHash - Hash generated by host
     */
    function registerChange(changeSetId, changeId, editScriptHash) {
        const key = `${changeSetId}:${changeId}`;
        changeHashes.set(key, editScriptHash);
        console.log('[ApprovalManager] Registered:', key);
    }

    /**
     * Accept a single change
     * @param {string} changeSetId
     * @param {string} changeId
     */
    function acceptChange(changeSetId, changeId) {
        const key = `${changeSetId}:${changeId}`;
        const hash = changeHashes.get(key);

        if (!hash) {
            console.error('[ApprovalManager] No hash for change:', key);
            showError('Unable to approve - missing verification data');
            return;
        }

        // Update UI immediately for responsiveness
        updateChangeStatus(changeId, 'applying');

        // Send approval with hash
        vscode.postMessage({
            type: 'changes:accept',
            payload: {
                changeSetId,
                changeId,
                editScriptHash: hash  // GAP-05: Echo back the hash
            }
        });

        console.log('[ApprovalManager] Accepted:', key);
    }

    /**
     * Reject a single change
     * @param {string} changeSetId
     * @param {string} changeId
     */
    function rejectChange(changeSetId, changeId) {
        // Update UI
        updateChangeStatus(changeId, 'rejected');

        // Send rejection (no hash needed for rejection)
        vscode.postMessage({
            type: 'changes:reject',
            payload: {
                changeSetId,
                changeId
            }
        });

        console.log('[ApprovalManager] Rejected:', `${changeSetId}:${changeId}`);
    }

    /**
     * Accept all changes in a change set
     * @param {string} changeSetId
     */
    function acceptAll(changeSetId) {
        // Get all pending changes
        const pendingChanges = window.qicState?.selectors.getPendingChanges();
        if (!pendingChanges || pendingChanges.id !== changeSetId) {
            console.error('[ApprovalManager] Change set not found:', changeSetId);
            return;
        }

        // Update UI for all
        pendingChanges.changes.forEach(change => {
            if (change.status === 'pending') {
                updateChangeStatus(change.id, 'applying');
            }
        });

        // Send accept all
        vscode.postMessage({
            type: 'changes:acceptAll',
            payload: { changeSetId }
        });

        console.log('[ApprovalManager] Accepted all in:', changeSetId);
    }

    /**
     * Reject all changes in a change set
     * @param {string} changeSetId
     */
    function rejectAll(changeSetId) {
        const pendingChanges = window.qicState?.selectors.getPendingChanges();
        if (!pendingChanges || pendingChanges.id !== changeSetId) {
            return;
        }

        // Update UI for all
        pendingChanges.changes.forEach(change => {
            if (change.status === 'pending') {
                updateChangeStatus(change.id, 'rejected');
            }
        });

        // Send reject all
        vscode.postMessage({
            type: 'changes:rejectAll',
            payload: { changeSetId }
        });

        console.log('[ApprovalManager] Rejected all in:', changeSetId);
    }

    /**
     * Retry failed changes
     * @param {string} changeSetId
     */
    function retryFailed(changeSetId) {
        const pendingChanges = window.qicState?.selectors.getPendingChanges();
        if (!pendingChanges) return;

        // Reset failed changes to pending in UI
        pendingChanges.changes.forEach(change => {
            if (change.status === 'failed') {
                updateChangeStatus(change.id, 'pending');
            }
        });

        // Send retry
        vscode.postMessage({
            type: 'changes:retryFailed',
            payload: { changeSetId }
        });
    }

    // ═══════════════════════════════════════════════════════════════════
    // UI Updates
    // ═══════════════════════════════════════════════════════════════════

    /**
     * Update change card status in UI
     * @param {string} changeId
     * @param {string} status
     */
    function updateChangeStatus(changeId, status) {
        const card = document.querySelector(`.qic-change-card[data-id="${changeId}"]`);
        if (!card) return;

        // Update data attribute
        card.dataset.status = status;

        // Update status indicator
        const indicator = card.querySelector('.qic-change-status');
        if (indicator) {
            indicator.textContent = getStatusText(status);
            indicator.className = `qic-change-status qic-change-status-${status}`;
        }

        // Update buttons
        const actions = card.querySelector('.qic-change-actions');
        if (actions) {
            const acceptBtn = actions.querySelector('[data-action="accept"]');
            const rejectBtn = actions.querySelector('[data-action="reject"]');

            if (status === 'applying' || status === 'applied') {
                acceptBtn?.setAttribute('disabled', 'true');
                rejectBtn?.setAttribute('disabled', 'true');
            } else if (status === 'rejected') {
                acceptBtn?.setAttribute('disabled', 'true');
                rejectBtn?.setAttribute('disabled', 'true');
            } else if (status === 'failed') {
                acceptBtn?.removeAttribute('disabled');
                rejectBtn?.setAttribute('disabled', 'true');
            }
        }
    }

    function getStatusText(status) {
        const texts = {
            'pending': 'Pending',
            'applying': 'Applying...',
            'applied': 'Applied',
            'rejected': 'Rejected',
            'failed': 'Failed'
        };
        return texts[status] || status;
    }

    /**
     * Handle change status update from host
     * @param {Object} payload
     */
    function handleStatusUpdate(payload) {
        const { changeSetId, changeId, status, error } = payload;

        updateChangeStatus(changeId, status);

        if (error) {
            showChangeError(changeId, error);
        }

        // Cleanup hash on completion
        if (status === 'applied' || status === 'rejected') {
            const key = `${changeSetId}:${changeId}`;
            changeHashes.delete(key);
        }
    }

    function showChangeError(changeId, error) {
        const card = document.querySelector(`.qic-change-card[data-id="${changeId}"]`);
        if (!card) return;

        let errorEl = card.querySelector('.qic-change-error');
        if (!errorEl) {
            errorEl = document.createElement('div');
            errorEl.className = 'qic-change-error';
            card.appendChild(errorEl);
        }

        errorEl.textContent = error;
    }

    function showError(message) {
        // Use toast or other notification
        if (window.qicInput?.showToast) {
            window.qicInput.showToast(message);
        } else {
            console.error('[ApprovalManager]', message);
        }
    }

    // ═══════════════════════════════════════════════════════════════════
    // Event Handling
    // ═══════════════════════════════════════════════════════════════════

    /**
     * Initialize event delegation for change cards
     */
    function initEventDelegation() {
        document.addEventListener('click', (e) => {
            const target = e.target;

            // Accept button
            if (target.closest('[data-action="accept"]')) {
                const card = target.closest('.qic-change-card');
                if (card) {
                    const changeSetId = card.dataset.changeSetId;
                    const changeId = card.dataset.id;
                    acceptChange(changeSetId, changeId);
                }
                return;
            }

            // Reject button
            if (target.closest('[data-action="reject"]')) {
                const card = target.closest('.qic-change-card');
                if (card) {
                    const changeSetId = card.dataset.changeSetId;
                    const changeId = card.dataset.id;
                    rejectChange(changeSetId, changeId);
                }
                return;
            }

            // Accept all button
            if (target.closest('[data-action="accept-all"]')) {
                const container = target.closest('.qic-changes-container');
                const changeSetId = container?.dataset.changeSetId;
                if (changeSetId) acceptAll(changeSetId);
                return;
            }

            // Reject all button
            if (target.closest('[data-action="reject-all"]')) {
                const container = target.closest('.qic-changes-container');
                const changeSetId = container?.dataset.changeSetId;
                if (changeSetId) rejectAll(changeSetId);
                return;
            }

            // Retry failed button
            if (target.closest('[data-action="retry-failed"]')) {
                const container = target.closest('.qic-changes-container');
                const changeSetId = container?.dataset.changeSetId;
                if (changeSetId) retryFailed(changeSetId);
                return;
            }
        });
    }

    // ═══════════════════════════════════════════════════════════════════
    // Message Handling
    // ═══════════════════════════════════════════════════════════════════

    /**
     * Handle incoming messages related to changes
     */
    function handleMessage(msg) {
        switch (msg.type) {
            case 'diff-preview':
            case 'changes:pending':
                // Register hashes for all changes
                if (msg.payload?.changes) {
                    msg.payload.changes.forEach(change => {
                        if (change.editScriptHash) {
                            registerChange(msg.payload.id, change.id, change.editScriptHash);
                        }
                    });
                }
                return true;

            case 'changes:fileStatus':
                handleStatusUpdate(msg.payload);
                return true;

            case 'changes:resolved':
                // Clear all hashes for this change set
                const prefix = `${msg.payload.id}:`;
                for (const key of changeHashes.keys()) {
                    if (key.startsWith(prefix)) {
                        changeHashes.delete(key);
                    }
                }
                return true;

            default:
                return false;
        }
    }

    // ═══════════════════════════════════════════════════════════════════
    // Export
    // ═══════════════════════════════════════════════════════════════════

    window.qicApproval = {
        registerChange,
        acceptChange,
        rejectChange,
        acceptAll,
        rejectAll,
        retryFailed,
        handleMessage,

        _debug: {
            getHashes: () => new Map(changeHashes),
            getExpandedId: () => expandedChangeId
        }
    };

    // Initialize on load
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initEventDelegation);
    } else {
        initEventDelegation();
    }

    console.log('[ApprovalManager] Initialized');

})();
```
